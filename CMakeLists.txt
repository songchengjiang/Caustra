cmake_minimum_required(VERSION 3.24)
project(Caustra VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CAUSTRA_BUILD_TESTS "Build CPU tests and GPU integration tests" ON)
option(CAUSTRA_GPU_TESTS "Register Vulkan GPU integration test with CTest" ON)

if(WIN32 AND NOT DEFINED ENV{VULKAN_SDK} AND NOT Vulkan_ROOT)
  file(GLOB vulkan_sdks LIST_DIRECTORIES true "C:/VulkanSDK/*")
  list(SORT vulkan_sdks COMPARE NATURAL ORDER DESCENDING)
  foreach(sdk IN LISTS vulkan_sdks)
    if(EXISTS "${sdk}/Lib/vulkan-1.lib")
      set(Vulkan_ROOT "${sdk}")
      break()
    endif()
  endforeach()
endif()
find_package(Vulkan 1.2 REQUIRED)
set(CAUSTRA_SLANG_ROOT "" CACHE PATH "Slang distribution directory (contains bin/slangc)")
find_program(CAUSTRA_SLANGC NAMES slangc
  HINTS "${CAUSTRA_SLANG_ROOT}/bin" "$ENV{SLANG_ROOT}/bin" "$ENV{VULKAN_SDK}/Bin"
  PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../vk_gltf_renderer/build/_deps/Slang-windows-x86_64-2026.13/bin"
        "${CMAKE_CURRENT_SOURCE_DIR}/../Formark/AuroraExternals/Slang/bin")
if(NOT CAUSTRA_SLANGC)
  message(FATAL_ERROR "Slang compiler not found. Set CAUSTRA_SLANG_ROOT to your Slang SDK or CAUSTRA_SLANGC to slangc.")
endif()
message(STATUS "Slang compiler: ${CAUSTRA_SLANGC}")

set(shader_spv "${CMAKE_CURRENT_BINARY_DIR}/generated/pathtrace.spv")
set(shader_header "${CMAKE_CURRENT_BINARY_DIR}/generated/pathtrace_spv.hpp")
file(GLOB shader_dependencies CONFIGURE_DEPENDS "third_party/nvshaders/*" "shaders/*.slang")
add_custom_command(OUTPUT "${shader_spv}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/generated"
  COMMAND "${CAUSTRA_SLANGC}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/pathtrace.slang"
    -target spirv -profile spirv_1_5 -entry main -stage compute -matrix-layout-row-major
    -I "${CMAKE_CURRENT_SOURCE_DIR}/third_party" -capability spvRayQueryKHR -O2 -g0 -o "${shader_spv}"
  DEPENDS ${shader_dependencies} VERBATIM COMMENT "Compiling Vulkan Ray Query path tracer")
add_custom_command(OUTPUT "${shader_header}"
  COMMAND ${CMAKE_COMMAND} "-DINPUT=${shader_spv}" "-DOUTPUT=${shader_header}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbedSpirv.cmake"
  DEPENDS "${shader_spv}" "cmake/EmbedSpirv.cmake" VERBATIM)

add_library(caustra STATIC src/renderer.cpp src/scene.cpp src/image.cpp "${shader_header}")
add_library(caustra::caustra ALIAS caustra)
target_include_directories(caustra PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(caustra PRIVATE Vulkan::Vulkan)
include(cmake/Dlss.cmake)
if(MSVC)
  # Float4 alignment intentionally introduces padding in host settings structs.
  target_compile_options(caustra PRIVATE /W4 /wd4324 /permissive- /utf-8)
else()
  target_compile_options(caustra PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
endif()

add_executable(caustra_demo examples/main.cpp examples/demo_scene.cpp)
target_link_libraries(caustra_demo PRIVATE caustra)
caustra_copy_dlss_runtime(caustra_demo)
set_target_properties(caustra_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
if(WIN32)
  target_link_libraries(caustra_demo PRIVATE user32 gdi32)
endif()
if(MSVC)
  target_compile_options(caustra_demo PRIVATE /utf-8)
endif()

if(CAUSTRA_BUILD_TESTS)
  enable_testing()
  add_executable(caustra_scene_tests tests/scene_tests.cpp src/scene.cpp src/image.cpp examples/demo_scene.cpp)
  target_include_directories(caustra_scene_tests PRIVATE include examples)
  if(MSVC)
    target_compile_options(caustra_scene_tests PRIVATE /utf-8)
  endif()
  add_test(NAME scene_contract COMMAND caustra_scene_tests)
  add_executable(caustra_renderer_tests tests/renderer_tests.cpp examples/demo_scene.cpp)
  target_include_directories(caustra_renderer_tests PRIVATE examples)
  target_link_libraries(caustra_renderer_tests PRIVATE caustra)
  caustra_copy_dlss_runtime(caustra_renderer_tests)
  if(CAUSTRA_GPU_TESTS)
    add_test(NAME renderer_gpu COMMAND caustra_renderer_tests --validation)
    set_tests_properties(renderer_gpu PROPERTIES LABELS GPU TIMEOUT 120 ENVIRONMENT "VK_LAYER_VALIDATE_SYNC=1")
  endif()
  if(CAUSTRA_ENABLE_DLSS)
    add_executable(caustra_dlss_tests tests/dlss_tests.cpp examples/demo_scene.cpp)
    target_include_directories(caustra_dlss_tests PRIVATE examples)
    target_link_libraries(caustra_dlss_tests PRIVATE caustra)
    caustra_copy_dlss_runtime(caustra_dlss_tests)
    option(CAUSTRA_DLSS_GPU_TESTS "Register DLSS integration test (requires NVIDIA RTX and NGX runtime)" OFF)
    if(CAUSTRA_GPU_TESTS AND CAUSTRA_DLSS_GPU_TESTS)
      add_test(NAME dlss_rr_gpu COMMAND caustra_dlss_tests)
      set_tests_properties(dlss_rr_gpu PROPERTIES LABELS "GPU;DLSS" TIMEOUT 180 ENVIRONMENT "VK_LAYER_VALIDATE_SYNC=1")
    endif()
  endif()
endif()
